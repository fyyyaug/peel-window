#include <metal_stdlib>
using namespace metal;

struct PeelVertex {
    float3 position;
    float2 textureCoordinate;
    float curl;
};

struct RasterData {
    float4 position [[position]];
    float2 textureCoordinate;
    float curl;
};

vertex RasterData peelVertex(
    uint vertexID [[vertex_id]],
    const device PeelVertex *vertices [[buffer(0)]]
) {
    PeelVertex input = vertices[vertexID];
    RasterData output;

    output.position = float4(input.position.xy, 0.0, 1.0);
    output.textureCoordinate = input.textureCoordinate;
    output.curl = input.curl;
    return output;
}

fragment float4 peelFragment(RasterData input [[stage_in]]) {
    float3 topColor = float3(0.91, 0.94, 1.00);
    float3 bottomColor = float3(0.58, 0.68, 0.92);
    float3 color = mix(bottomColor, topColor, input.textureCoordinate.y);

    float titleBar = smoothstep(0.83, 0.86, input.textureCoordinate.y);
    color = mix(color, float3(0.16, 0.20, 0.31), titleBar);

    float edgeX = min(input.textureCoordinate.x, 1.0 - input.textureCoordinate.x);
    float edgeY = min(input.textureCoordinate.y, 1.0 - input.textureCoordinate.y);
    float edge = smoothstep(0.0, 0.018, min(edgeX, edgeY));
    color *= mix(0.68, 1.0, edge);

    float backside = smoothstep(0.49, 0.51, input.curl);
    color = mix(color, float3(0.88, 0.84, 0.76), backside);
    float light = 1.0 - 0.16 * sin(input.curl * 3.14159265);
    return float4(color * light, 1.0);
}

fragment float4 peelSnapshotFragment(
    RasterData input [[stage_in]],
    texture2d<float> snapshot [[texture(0)]]
) {
    constexpr sampler imageSampler(filter::linear, address::clamp_to_edge);
    float3 color = snapshot.sample(imageSampler, input.textureCoordinate).rgb;
    float backside = smoothstep(0.49, 0.51, input.curl);
    color = mix(color, float3(0.88, 0.84, 0.76), backside);
    float light = 1.0 - 0.16 * sin(input.curl * 3.14159265);
    return float4(color * light, 1.0);
}

fragment float4 peelUnderlayFragment(
    RasterData input [[stage_in]],
    texture2d<float> snapshot [[texture(0)]],
    constant float4 &uvMapping [[buffer(0)]]
) {
    constexpr sampler imageSampler(filter::linear, address::clamp_to_edge);
    float2 coordinates = input.textureCoordinate * uvMapping.xy + uvMapping.zw;
    return snapshot.sample(imageSampler, coordinates);
}
